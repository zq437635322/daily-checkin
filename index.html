配置云端存储（Supabase）

    
    
    
    
    

      
      
    

    
数据存云端、跨设备同步。anon key 设计上可放前端（配合行级安全策略）。

      首次使用：请在 Supabase 的 SQL Editor 执行下面这段建表语句（可整段复制）：
      
create table if not exists public.checkin_state (
  id text primary key,
  data jsonb,
  updated_at timestamptz default now()
);
alter table public.checkin_state enable row level security;
drop policy if exists "anon_all" on public.checkin_state;
create policy "anon_all" on public.checkin_state
  for all to anon using (true) with check (true);
